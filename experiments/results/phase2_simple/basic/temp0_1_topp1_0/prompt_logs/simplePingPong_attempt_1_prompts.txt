  private var count = 0

  def receive: Receive = {
    case StartMessage =>
      println("Ping started the game!")
      pong ! PingMessage

    case PongMessage =>
      count += 1
      println(s"Ping received pong #$count from ${sender().path.name}")

        println("Ping stopped")
        context.stop(self)

      }
  }
}
