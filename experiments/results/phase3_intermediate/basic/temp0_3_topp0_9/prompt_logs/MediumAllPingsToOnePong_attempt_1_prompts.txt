import akka.actor._

class PingActor(pong: ActorRef, id: Int) extends Actor {
  def receive: Receive = {
    case "start" =>
      pong ! s"ping-$id"

    case msg: String =>
      pong ! s"ping-$id"
  }
}

class PongActor extends Actor {
  def receive: Receive = {
    case msg: String =>
      sender() ! s"pong-for-$msg"
  }
}
