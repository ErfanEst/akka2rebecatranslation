package MediumAllPingsToOnePong

import akka.actor._

class PingActor(pong: ActorRef, id: Int) extends Actor {
  def receive: Receive = {
    case "start" =>
      pong ! s"ping-$id"

    case msg: String =>
      pong ! s"ping-$id"
  }
}

class PongActor extends Actor {
  def receive: Receive = {
    case msg: String =>
      sender() ! s"pong-for-$msg"
  }
}

object AllPingsToOnePongApp extends App {
  val system = ActorSystem("AllToOneSystem")
  val pong = system.actorOf(Props[PongActor], "pong")

  (1 to 10).foreach { i =>
    val ping = system.actorOf(Props(new PingActor(pong, i)), s"ping$i")
    ping ! "start"
  }
}
