case object Ping
case object Pong
case class Register(ref: ActorRef)

    case "start" =>
      hub ! Ping
    case Pong =>
      hub ! Ping

class Hub extends Actor {
  var clients: List[ActorRef] = Nil

  def receive: Receive = {
    case Register(ref) =>
      clients ::= ref
    case Ping =>
      sender() ! Pong
  }
}
