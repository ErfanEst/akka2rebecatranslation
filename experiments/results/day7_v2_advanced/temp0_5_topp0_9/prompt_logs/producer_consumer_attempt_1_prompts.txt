package ProducerConsumer

import akka.actor._
import scala.collection.mutable.Queue

case class Produce(item: String)
case class Consume()
case class Item(data: String)

class Producer(consumer: ActorRef) extends Actor {
  def receive: Receive = {
    case Produce(item) =>
      println(s"Producer: sending $item")
      consumer ! Item(item)
  }
}

class Consumer extends Actor {
  private val buffer = Queue[String]()
  
  def receive: Receive = {
    case Item(data) =>
      buffer.enqueue(data)
      println(s"Consumer: received $data, buffer size: ${buffer.size}")
      
    case Consume() =>
      if (buffer.nonEmpty) {
        val item = buffer.dequeue()
        println(s"Consumer: consumed $item")
      } else {
        println("Consumer: buffer empty")
      }
  }
}

object ProducerConsumerSystem extends App {
  val system = ActorSystem("ProducerConsumerSystem")
  val consumer = system.actorOf(Props[Consumer], "consumer")
  val producer = system.actorOf(Props(new Producer(consumer)), "producer")
  
  producer ! Produce("item1")
  producer ! Produce("item2")
  consumer ! Consume()
}
