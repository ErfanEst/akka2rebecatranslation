package SimpleExample

import akka.actor._

case object PingMessage
case object PongMessage
case object StartMessage
case object StopMessage

class Ping(pong: ActorRef) extends Actor {
  private var count = 0

  def receive: Receive = {
    case StartMessage =>
      println("Ping started the game!")
      pong ! PingMessage

    case PongMessage =>
      count += 1
      println(s"Ping received pong #$count from ${sender().path.name}")
      if (count >= 10) {
        sender() ! StopMessage
        println("Ping stopped")
        context.stop(self)
      } else {
        sender() ! PingMessage
      }
  }
}

class Pong extends Actor {
  def receive: Receive = {
    case PingMessage =>
      println(s"Pong received ping from ${sender().path.name}")
      sender() ! PongMessage
    case StopMessage =>
      println("Pong stopped")
      context.stop(self)
  }
}

object PingPong extends App {
  private val system = ActorSystem("PingPongSystem")
  private val pong = system.actorOf(Props[Pong], name = "pong")
  private val ping = system.actorOf(Props(new Ping(pong)), name = "ping")

  // ✅ Start the exchange from inside an actor (ping)

Rebeca does not have a direct equivalent to Akka's actor model. However, the same logic can be implemented using reactive classes (Rebec) in Rebeca. Here is a Rebeca equivalent for the given Akka code:

```
reactiveclass Ping(1) {
    knownrebecs { Pong pong; }
    statevars { int count = 0; }

    msgsrv StartMessage {
        println("Ping started the game!");
        pong.PingMessage();
    }

    msgsrv PongMessage {
        count = count + 1;
        println("Ping received pong #" + count + " from Pong");
        if (count >= 10) {
            pong.StopMessage();
            println("Ping stopped");
            selfdestruct();
        } else {
            pong.PingMessage();
        }
    }
}

reactiveclass Pong(1) {
    knownrebecs { Ping ping; }

    msgsrv PingMessage {
        println("Pong received ping from Ping");
        ping.PongMessage();
    }

    msgsrv StopMessage {
        println("Pong stopped");
        selfdestruct();
    }
}

main {
    Pong pong( );
    Ping ping(pong);
    ping.StartMessage();
}
```

Please note that the `selfdestruct()` method in Rebeca is used to stop the execution of a reactive class, which is equivalent to `context.stop(self)` in Akka. Also, the `knownrebecs` block is used to declare other reactive classes that this reactive class can send messages to.
