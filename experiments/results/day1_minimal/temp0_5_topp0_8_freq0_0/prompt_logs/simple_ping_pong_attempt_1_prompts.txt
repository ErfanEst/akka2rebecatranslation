package SimpleExample

import akka.actor._

case object PingMessage
case object PongMessage
case object StartMessage
case object StopMessage

class Ping(pong: ActorRef) extends Actor {
  private var count = 0

  def receive: Receive = {
    case StartMessage =>
      println("Ping started the game!")
      pong ! PingMessage

    case PongMessage =>
      count += 1
      println(s"Ping received pong #$count from ${sender().path.name}")
      if (count >= 10) {
        sender() ! StopMessage
        println("Ping stopped")
        context.stop(self)
      } else {
        sender() ! PingMessage
      }
  }
}

class Pong extends Actor {
  def receive: Receive = {
    case PingMessage =>
      println(s"Pong received ping from ${sender().path.name}")
      sender() ! PongMessage
    case StopMessage =>
      println("Pong stopped")
      context.stop(self)
  }
}

object PingPong extends App {
  private val system = ActorSystem("PingPongSystem")
  private val pong = system.actorOf(Props[Pong], name = "pong")
  private val ping = system.actorOf(Props(new Ping(pong)), name = "ping")

  // ✅ Start the exchange from inside an actor (ping)
  ping ! StartMessage
}

Please note that in Rebeca, we don't have a direct equivalent to Akka's `sender()`. Instead, we declare known rebecs (other reactive classes that this reactive class can send messages to) in the `knownrebecs` section of each reactive class. 

In addition, Rebeca does not support string concatenation using the `+` operator. You can use it here as a pseudo code.

Also, the `selfdestruct()` function in Rebeca is used to stop the current rebec (similar to `context.stop(self)` in Akka).
