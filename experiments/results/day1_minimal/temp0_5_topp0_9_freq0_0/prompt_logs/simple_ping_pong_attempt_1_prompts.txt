package SimpleExample

import akka.actor._

case object PingMessage
case object PongMessage
case object StartMessage
case object StopMessage

class Ping(pong: ActorRef) extends Actor {
  private var count = 0

  def receive: Receive = {
    case StartMessage =>
      println("Ping started the game!")
      pong ! PingMessage

    case PongMessage =>
      count += 1
      println(s"Ping received pong #$count from ${sender().path.name}")
      if (count >= 10) {
        sender() ! StopMessage
        println("Ping stopped")
        context.stop(self)
      } else {
        sender() ! PingMessage
      }
  }
}

class Pong extends Actor {
  def receive: Receive = {
    case PingMessage =>
      println(s"Pong received ping from ${sender().path.name}")
      sender() ! PongMessage
    case StopMessage =>
      println("Pong stopped")
      context.stop(self)
  }
}

object PingPong extends App {
  private val system = ActorSystem("PingPongSystem")
  private val pong = system.actorOf(Props[Pong], name = "pong")
  private val ping = system.actorOf(Props(new Ping(pong)), name = "ping")

  // ✅ Start the exchange from inside an actor (ping)
  ping ! StartMessage
}

Please note that Rebeca does not support dynamic creation of rebecs (actors), so the `Props` mechanism in Akka has no equivalent in Rebeca. Also, Rebeca does not support the `sender()` mechanism in Akka, so the name of the sender of a message cannot be determined dynamically. The `context.stop(self)` in Akka is equivalent to `self.terminate()` in Rebeca.
