package ProducerConsumer

import akka.actor._
import scala.collection.mutable.Queue

case class Produce(item: String)
case class Consume()
case class Item(data: String)

class Producer(consumer: ActorRef) extends Actor {
  def receive: Receive = {
    case Produce(item) =>
      println(s"Producer: sending $item")
      consumer ! Item(item)
  }
}

class Consumer extends Actor {
  private val buffer = Queue[String]()
  
  def receive: Receive = {
    case Item(data) =>
      buffer.enqueue(data)
      println(s"Consumer: received $data, buffer size: ${buffer.size}")
      
    case Consume() =>
      if (buffer.nonEmpty) {
        val item = buffer.dequeue()
        println(s"Consumer: consumed $item")
      } else {
        println("Consumer: buffer empty")
      }
  }
}

object ProducerConsumerSystem extends App {
  val system = ActorSystem("ProducerConsumerSystem")
  val consumer = system.actorOf(Props[Consumer], "consumer")
  val producer = system.actorOf(Props(new Producer(consumer)), "producer")
  
  producer ! Produce("item1")
  producer ! Produce("item2")
  consumer ! Consume()
}

reactiveclass Consumer(5) {
	knownrebecs {
		// no known rebecs
	}

	statevars {
		int bufferSize;
		char buffer[10][20]; // simple fixed-size buffer of up to 10 items
	}

	Consumer() {
		bufferSize = 0;
	}

	msgsrv item(char data[20]) {
		// enqueue data if there is space
		if (bufferSize < 10) {
			int i;
			i = 0;
			while (i < 20) {
				buffer[bufferSize][i] = data[i];
				i = i + 1;
			}
			bufferSize = bufferSize + 1;
			// Consumer: received data, buffer size updated
		} else {
			// buffer full, item dropped
		}
	}

	msgsrv consume() {
		if (bufferSize > 0) {
			// dequeue first item by shifting
			int i;
			int j;
			i = 0;
			while (i < bufferSize - 1) {
				j = 0;
				while (j < 20) {
					buffer[i][j] = buffer[i + 1][j];
					j = j + 1;
				}
				i = i + 1;
			}
			bufferSize = bufferSize - 1;
			// Consumer: consumed one item
		} else {
			// Consumer: buffer empty
		}
	}
}
