package MediumInfinitePingPong

import akka.actor._

case object Ping
case object Pong
case object Start

class PingActor(pong: ActorRef) extends Actor {
  def receive: Receive = {
    case Start =>
      pong ! Ping
    case Pong =>
      pong ! Ping
  }
}

class PongActor extends Actor {
  def receive: Receive = {
    case Ping =>
      sender() ! Pong
  }
}

object InfiniteApp extends App {
  val system = ActorSystem("InfiniteSystem")
  val pong = system.actorOf(Props[PongActor], "pong")
  val ping = system.actorOf(Props(new PingActor(pong)), "ping")

  ping ! Start
}
