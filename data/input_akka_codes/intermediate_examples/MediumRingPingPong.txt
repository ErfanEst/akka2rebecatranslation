package MediumRingPingPong

import akka.actor._

case object Ping
case class SetNext(ref: ActorRef)

class RingActor extends Actor {
  var next: ActorRef = _

  def receive: Receive = {
    case SetNext(ref) =>
      next = ref

    case Ping =>
      next ! Ping
  }
}

object RingApp extends App {
  val system = ActorSystem("RingSystem")

  val actors = (1 to 5).map { i =>
    system.actorOf(Props[RingActor], s"a$i")
  }

  actors.zip(actors.tail :+ actors.head).foreach {
    case (a, b) => a ! SetNext(b)
  }

  actors.head ! Ping
}
